2. Row-Based Canvas Logic (enhance existing Canvas.jsx)
javascript// Add row intelligence to existing Canvas.jsx
const handleRowBasedDrop = (item, dropPosition) => {
  const { row, column } = dropPosition;
  
  // Calculate width based on how many widgets will be in this row
  const existingInRow = widgets.filter(w => w.position.y === row);
  const newColumnCount = existingInRow.length + 1;
  
  let widgetWidth;
  if (newColumnCount === 1) widgetWidth = 12; // Full width
  else if (newColumnCount === 2) widgetWidth = 6; // Half width  
  else if (newColumnCount === 3) widgetWidth = 4; // Third width
  else widgetWidth = 3; // Quarter width
  
  // Resize existing widgets in the row
  existingInRow.forEach((widget, index) => {
    updateWidgetPosition(widget.id, {
      ...widget.position,
      x: index * widgetWidth,
      w: widgetWidth
    });
  });
  
  // Add new widget
  addWidget(item.type, {
    x: existingInRow.length * widgetWidth,
    y: row,
    w: widgetWidth,
    h: item.defaultSize.h
  });
};
